cmake_minimum_required(VERSION 3.20)
project(VoxelForge VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(VOXELFORGE_NATIVE "Optimize for this CPU (off: x86-64-v3 for portable builds)" ON)
add_compile_options(-Wall -Wextra -Wno-unused-parameter)
add_compile_definitions(VOXELFORGE_VERSION="${PROJECT_VERSION}")
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  add_compile_options(-O3 -ffast-math)
  if(VOXELFORGE_NATIVE)
    add_compile_options(-march=native)
  else()
    add_compile_options(-march=x86-64-v3)
  endif()
endif()
find_package(OpenMP REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(OpenGL REQUIRED COMPONENTS OpenGL EGL)
pkg_check_modules(TURBOJPEG REQUIRED IMPORTED_TARGET libturbojpeg)
add_library(voxelforge_core STATIC src/device.cpp src/stereo.cpp src/tsdf.cpp src/mesh.cpp src/meshedit.cpp src/project.cpp src/register.cpp src/markers.cpp src/measure.cpp src/pipeline.cpp src/source.cpp src/color.cpp src/settings.cpp src/gpustereo.cpp)
target_include_directories(voxelforge_core PUBLIC src)
target_link_libraries(voxelforge_core PUBLIC OpenMP::OpenMP_CXX PkgConfig::TURBOJPEG OpenGL::OpenGL OpenGL::EGL)
add_executable(voxelforge-cli src/cli.cpp)
target_link_libraries(voxelforge-cli voxelforge_core)
add_executable(voxelforge-tests tests/tests.cpp)
target_link_libraries(voxelforge-tests voxelforge_core)

option(VOXELFORGE_GUI "Build the graphical application" ON)
if(VOXELFORGE_GUI)
  include(FetchContent)
  FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.9b GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(imgui)
  find_package(glfw3 REQUIRED)
  find_package(OpenGL REQUIRED COMPONENTS OpenGL)
  pkg_check_modules(PNG REQUIRED IMPORTED_TARGET libpng)
  add_executable(voxelforge src/gui.cpp src/editui.cpp src/meshview.cpp src/projectui.cpp src/theme.cpp src/filedialog.cpp src/slicers.cpp src/homeui.cpp
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(voxelforge PRIVATE ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_link_libraries(voxelforge voxelforge_core glfw OpenGL::OpenGL PkgConfig::PNG)
  install(TARGETS voxelforge voxelforge-cli DESTINATION bin)
endif()
